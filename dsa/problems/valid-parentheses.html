31. Valid Parentheses

EasyStacks, Queues & Monotonic Stacks✓ SolvedAlso on LeetCode ↗

s contains only the characters ()[]{}. It is valid when every opening bracket is closed by the same type of bracket, in the correct order.

Example 1

Input: s = "()[]{}"
Output: true

Example 2

Input: s = "(]"
Output: false
Hint 1

The most recent unclosed bracket must be closed first: last in, first out.

StackString

Approach 1: Stack ✅

Push openers; on a closer, pop and check the type.

⏱ Time O(n)💾 Space O(n)

Reference solution for Stack (time O(n), space O(n)). Each solution passes every test in all four languages.

class Solution:
    def isValid(self, s: str) -> bool:
        pair = {')': '(', ']': '[', '}': '{'}
        stack = []
        for ch in s:
            if ch in pair:
                if not stack or stack.pop() != pair[ch]:
                    return False          # wrong or missing opener
            else:
                stack.append(ch)
        return not stack                  # nothing left unclosed
class Solution {
    public boolean isValid(String s) {
        Deque<Character> stack = new ArrayDeque<>();
        for (char ch : s.toCharArray()) {
            if (ch == '(' || ch == '[' || ch == '{') stack.push(ch);
            else {
                if (stack.isEmpty()) return false;
                char open = stack.pop();
                if ((ch == ')' && open != '(') || (ch == ']' && open != '[') || (ch == '}' && open != '{')) return false;
            }
        }
        return stack.isEmpty();
    }
}
class Solution {
public:
    bool isValid(string s) {
        stack<char> st;
        for (char ch : s) {
            if (ch == '(' || ch == '[' || ch == '{') st.push(ch);
            else {
                if (st.empty()) return false;
                char open = st.top(); st.pop();
                if ((ch == ')' && open != '(') || (ch == ']' && open != '[') || (ch == '}' && open != '{')) return false;
            }
        }
        return st.empty();
    }
};
function isValid(s) {
  const pair = { ')': '(', ']': '[', '}': '{' }, stack = [];
  for (const ch of s) {
    if (ch in pair) { if (stack.pop() !== pair[ch]) return false; }
    else stack.push(ch);
  }
  return stack.length === 0;
}
